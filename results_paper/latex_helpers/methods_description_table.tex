% Brief description of the discretisation methods compared in the
% one-dimensional experiments (see src/eig_*.m).  The method labels are the
% ones used in the DOF-sweep figures and the eigenvalue comparison tables.

\begin{table}
  \centering
  \begin{tabular}{l p{0.72\linewidth}}
    \toprule
    Discretisation method & Description \\
    \midrule
    \texttt{DST} & Discrete sine transform method as described in this work, see Algorithm~\ref{alg:dst-line}. \\
    \texttt{FD} & Finite difference method. Standard second-order three-point stencil on the uniform interior grid of $N$ points, the potential evaluated at the grid points, own implementation. \\
    \texttt{CDM} & Chebyshev differentiation matrix method. Second-derivative differentiation matrix at $N+2$ Chebyshev points of the second kind as implemented in the \texttt{chebfun} library, see~\cite{driscoll.ta.hale.n.ea:chebfun}, \lstinline{diffmat} function; the two boundary rows and columns are removed to impose the homogeneous Dirichlet conditions, see also \cite{trefethen.ln:spectral}. The potential evaluated at the grid points, own implementation. \\
    \texttt{MBCDM} & Mapped barycentric Chebyshev differentiation matrix method, \cite{zhang.x:mapped}. As \texttt{CDM}, with the grid pushed towards a uniform distribution by the Kosloff--Tal-Ezer map $g(s) = \frac{\arcsin(\alpha s)}{\arcsin \alpha}$, $\alpha \in (0, 1]$, which relaxes the clustering of the Chebyshev points at the endpoints, \cite{kosloff.d.tal-ezer.h:modified}; the differentiation matrix is transformed by the chain rule, own implementation on top of \texttt{chebfun} library. The value $\alpha = 0.999$ is used throughout. \\
    \texttt{LGCC} & Legendre--Galerkin method. Galerkin discretisation on $[-1, 1]$ in the compact basis $\varphi_k = L_k - L_{k+2}$, $L_k$ being Legendre polynomials, see~\cite{shen.j.tang.t.ea:spectral}; the potential matrix is assembled from the Legendre coefficients of $q \varphi_j$ computed with the \texttt{chebfun} library, see~\cite{driscoll.ta.hale.n.ea:chebfun}, \lstinline{legcoeffs}/\lstinline{legpoly} functions. Own implementation. \\ % Collocation variant is discussed in \cite{chen.l.ma.h:approximate}
    \midrule
    \texttt{Numerov} & Numerov method\tablefootnote{The method does not provide a straightforward discretisation of the corresponding operator at the continuous level. The method is designed as a method for solving boundary value problems and/or for computing the eigenvalues as it ``needs the right-hand side''.}, originally~\cite{noumerov.bv:method} and \cite{numerov.b:note}, see also \cite{hairer.e.nrsett.sp.ea:solving} and \cite{andrew.al:asymptotic}. Fourth-order three-point scheme on the uniform interior grid of $N$ points. Own implementation. \\
    \texttt{Numerov+AC} & Numerov method \texttt{Numerov} with the asymptotic correction\tablefootnote{The asymptotic correction improves computation of the eigenvalues by the standard Numerov method. Consequently the method does not provide a discretisation of the corresponding operator at the continuous level.}, see \cite{andrew.al:asymptotic}. The $k$-th \texttt{Numerov} eigenvalue is shifted by the error the scheme commits on the same problem with $q = 0$, see~\cite{paine.jw.hoog.fr.ea:on}, which removes the leading discretisation error of the higher index eigenvalues. Own implementation. \\
    \bottomrule
  \end{tabular}
  \caption{Discretisation of the Sturm--Liouville operator $-y'' + q(x) y$ with zero Dirichlet boundary conditions.}
  \label{tab:methods_description_line}
\end{table}
